% --- макроси верстки поетичного тексту --------------------------------
\newcommand{\poemchapter}[1]{%
  \clearpage
  \begin{center}
    {\sffamily\large Розділ #1}\par
  \end{center}
  \vspace{4mm}}

\newcommand{\poemverse}[1]{%
  \vspace{3mm}%
  {\sffamily\footnotesize #1}\par
  \vspace{1mm}}

% Одна строфа: рядки розділені `\\`, тому збираємо їх у звичайний абзац
% із збільшеним міжрядковим інтервалом.
\newcommand{\poemstanza}[1]{%
  \begingroup
  \setlength{\parindent}{0pt}%
  \setlength{\baselineskip}{1.45\baselineskip}%
  \begin{flushleft}#1\end{flushleft}%
  \endgroup
  \vspace{2mm}}

\newcommand{\poemsection}[1]{%
  \vspace{6mm}%
  {\sffamily\normalsize #1}\par
  \vspace{2mm}}

\newcommand{\poemnote}[1]{%
  \begingroup
  \small
  \begin{flushleft}#1\end{flushleft}%
  \endgroup
  \vspace{3mm}}
